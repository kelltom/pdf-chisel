**Conversion loop (Option A from research — renderer-side rendering, no Worker Thread):**
```typescript
async function execute() {
  if (!appState.currentFile) return
  appState.isProcessing = true
  isConverting = true
  progressCurrent = 0
  progressTotal = appState.currentFile.pageCount
  outputFiles = []
  outputFileNames = []
  conversionError = null

  try {
    const folder = await window.api.makeConvertOutputFolder()
    outputFolder = folder

    const pdfBytes = await window.api.readFileBytes(appState.currentFile.filePath)

    const totalPages = appState.currentFile.pageCount
    const padWidth = totalPages > 99 ? 3 : totalPages > 9 ? 2 : 1
    const ext = format === 'jpeg' ? 'jpg' : 'png'

    for (let i = 1; i <= totalPages; i++) {
      progressCurrent = i
      const dataUrl = await renderPageToDataUrl(new Uint8Array(pdfBytes), i, dpi, format)
      const fileName = `page-${String(i).padStart(padWidth, '0')}.${ext}`
      const filePath = await window.api.writeImageFile({ dataUrl, outputFolder: folder, fileName })
      outputFiles = [...outputFiles, filePath]
      outputFileNames = [...outputFileNames, fileName]
    }

    hasConversionResult = true
    // Auto-open output folder (matches Phase 2 ResultsSummary behavior)
    window.api.openOutputFolder(folder)
  } catch (err) {
    conversionError = {
      cause: err instanceof Error ? err.message : String(err),
      fix: 'Try a different PDF file or reduce the DPI setting.'
    }
  } finally {
    isConverting = false
    appState.isProcessing = false
  }
}
```

Note: `window.api.readFileBytes` returns a `Buffer` over IPC, which arrives as a plain object (not Uint8Array) in the renderer. Wrap it: `new Uint8Array(Object.values(pdfBytes as unknown as Record<string, number>))`. Alternatively, if pdfjs-dist's `getDocument({ data: buffer })` accepts a Buffer directly, omit the conversion. Test and use whichever works.
